\section{Motivation \& Background}
% =============================================================================

\begin{frame}{The Problem}
\begin{columns}[T]
\column{0.55\textwidth}
\textbf{How do we make LLMs respond with the right \emph{affective tone}?}

\bigskip

Current approaches:
\begin{itemize}
    \item \textbf{Prompt engineering} --- ``Reply warmly and supportively''
    \item \textbf{RLHF / fine-tuning} --- Requires extensive data
    \item \textbf{Neither} offers a principled, \emph{theory-grounded} model
          of affect
\end{itemize}

\column{0.42\textwidth}
\alert{Our approach:} Combine two frameworks:
\begin{enumerate}
    \item \textbf{Affect Control Theory} (ACT) ---
          a sociological model that predicts socially optimal affective
          behaviour given identity roles
    \item \textbf{Representation Engineering} (RepE) ---
          steers LLM outputs by manipulating internal activation directions
\end{enumerate}
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Affect Control Theory (ACT) Primer}
\begin{columns}[T]
\column{0.5\textwidth}
ACT models affect as a 3-D \textbf{EPA} space:
\begin{description}
    \item[Evaluation (E)] Good--Bad (valence)
    \item[Potency (P)] Strong--Weak (dominance)
    \item[Activity (A)] Active--Quiet (arousal)
\end{description}

\bigskip

Key concepts:
\begin{itemize}
    \item \textbf{Fundamental sentiments} --- stable EPA values for
          identities (``counselor'', ``boss'') and behaviours
    \item \textbf{Transient impressions} --- how identities are perceived
          after an interaction event
    \item \textbf{Deflection} --- the discrepancy between fundamental
          sentiments and transient impressions
\end{itemize}

\column{0.47\textwidth}
\textbf{Deflection minimisation:}

Given roles (agent identity, user identity) and a user's observed behaviour
EPA, ACT computes the \emph{optimal response EPA} that minimises deflection.

\bigskip

\textbf{Example:}\\
A \emph{counselor} responding to an angry \emph{client} $\rightarrow$
ACT predicts the counselor should respond with moderately high E
(warm), moderate P (assertive), and low A (calm).

\bigskip

This gives us a \emph{computable target} for each conversation turn.
\end{columns}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Representation Engineering (RepE) Primer}
\begin{columns}[T]
\column{0.52\textwidth}
RepE (\textit{Zou et al., 2023}) discovers \textbf{linear directions}
in activation space that correspond to semantic concepts.

\bigskip

\textbf{Our use:}
\begin{enumerate}
    \item \textbf{Reading:} Project activations onto E, P, A directions
          $\rightarrow$ continuous EPA values
    \item \textbf{Steering:} Add scaled direction vectors during generation
          $\rightarrow$ shift output EPA toward target
\end{enumerate}

\column{0.45\textwidth}
\textbf{Pipeline Overview:}

\begin{enumerate}
    \item User sends message $m$
    \item \textbf{Read} EPA of $m$ via RepE reader
    \item \textbf{ACT} computes optimal response EPA target
    \item \textbf{Steer} generation by injecting weighted EPA direction
          vectors into the model's hidden states
    \item \textbf{Re-read} generated response EPA to verify
\end{enumerate}

\bigskip

\textit{Model:} Llama-3.1-8B-Instruct\\
\textit{Layers:} 32 transformer layers, directions extracted across
layers $-31$ to $-1$
\end{columns}
\end{frame}

\section{System Architecture}
% =============================================================================

\begin{frame}{System Components}
\begin{columns}[T]
\column{0.48\textwidth}
\textbf{Core modules} (in \texttt{act\_three/}):

\begin{description}
    \item[\texttt{act\_core.py}] ACT impression-formation equations,
         deflection minimisation, EPA dataclass
    \item[\texttt{epa\_reader.py}] \texttt{EPAReader} --- reads EPA from
         text via calibrated RepE projections
    \item[\texttt{epa\_steerer.py}] \texttt{EPASteerer} --- steers
         generation by injecting activation perturbations
    \item[\texttt{epa\_calibration.py}] Calibration coefficients
         (linear/affine transform from raw RepE readings to EPA scale)
\end{description}

\column{0.48\textwidth}
\textbf{Data \& Config:}
\begin{description}
    \item[\texttt{prompt\_formatting.py}] Llama-3 chat template utilities
    \item[\texttt{tuning\_dataset.py}] Dataset generation, stratified splits
    \item[\texttt{identities.csv}] ACT dictionary of identity EPA values
\end{description}

\bigskip

\textbf{CLI Pipeline:}
\begin{description}
    \item[\texttt{extract\_directions.py}] Standalone direction extraction
    \item[\texttt{orthogonalise\_directions.py}] Gram--Schmidt E$\perp$P$\perp$A
    \item[\texttt{tune\_reader.py}] Reader tuning (layer selection + calibration)
\end{description}

\bigskip

\textbf{Experiment suite} (in \texttt{experiments/}):
\begin{itemize}
    \item 10 experiment scripts (\texttt{01}--\texttt{10})
    \item Statistical testing utilities (\texttt{stats\_utils.py})
    \item Shared config, setup, scenarios
    \item 12 automated figure types
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Exp 02: Steering Analysis}

\textbf{What works:}
\begin{itemize}
    \item \textcolor{AccentGreen}{Evaluation steering is effective}:
          57.4\% hit rate (above chance), mean distance improvement of
          0.130 on the EPA scale
    \item 85\% of trials show improvement on \emph{at least one} dimension
\end{itemize}

\bigskip

\textbf{What doesn't:}
\begin{itemize}
    \item \textcolor{AccentRed}{Potency steering actively harms}:
          25.4\% hit rate (below chance), mean distance \emph{increases}
          by 0.160
    \item This is consistent with the weak reading quality for P
          ($\rho = 0.57$) --- if the reader can't measure P well,
          steering toward a P target is poorly calibrated
    \item Activity is near chance (52.6\%), suggesting the direction
          vector captures some signal but steering magnitude is
          suboptimal
\end{itemize}

\bigskip

\textbf{Root cause:} The Potency direction may not be a clean linear
concept in Llama's activation space, or may be entangled with other
dimensions.
\end{frame}

\section{Improved Pipeline \& New Experiments}
% =============================================================================

% -----------------------------------------------------------------------------
\begin{frame}{Potency Direction Fix}

\textbf{Root cause:} The original extraction template ``Pretend you're a \{powerful/weak\} person'' conflated Potency with Evaluation (good$\approx$powerful overlap).

\bigskip

\textbf{Fixes applied:}
\begin{enumerate}
    \item \textbf{Improved extraction adjectives:}
    \begin{itemize}
        \item Before: ``a powerful'' / ``a weak''
        \item After: ``an authoritative and commanding'' / ``a meek and submissive''
    \end{itemize}

    \item \textbf{Enriched term sets:} Replaced valence-leaking terms
    (potent, powerful, strong $\rightarrow$ commanding, assertive, decisive, imposing)

    \item \textbf{Gram--Schmidt orthogonalisation:}
    Per-layer orthogonalisation ensures E$\perp$P$\perp$A.
    Priority order: E (unchanged) $\rightarrow$ P$\perp$E $\rightarrow$ A$\perp$\{E,P\}

    \item \textbf{Per-dimension coefficients:}
    E: 3.0 (from sweep peak), P: TBD, A: 2.5 (monotonic region)
\end{enumerate}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Exp 08: Hybrid Steering (PE + RepE)}

\textbf{Hypothesis:} Combining prompt engineering (better for P, A) with
RepE steering (better for E) should outperform either method alone.

\bigskip

\textbf{Four conditions per trial:}
\begin{enumerate}
    \item \textbf{Unsteered:} Neutral system prompt only
    \item \textbf{RepE only:} Neutral prompt + activation steering
    \item \textbf{PE only:} Affective system prompt, no steering
    \item \textbf{Hybrid:} Affective prompt + activation steering
\end{enumerate}

\bigskip

\textbf{Evaluation:}
\begin{itemize}
    \item Per-dimension distance to ACT target with bootstrap CIs
    \item Paired permutation tests (hybrid vs each baseline)
    \item Cohen's $d$ effect sizes
\end{itemize}

\bigskip

\textit{Expected result:} Hybrid should dominate on E (from RepE) and match
or beat PE on P and A (from the affective system prompt).
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Exp 09: Direction Quality Comparison}

\textbf{Purpose:} Quantify the improvement from direction re-extraction
and orthogonalisation.

\bigskip

\textbf{Compares three direction sets:}
\begin{enumerate}
    \item \textbf{Original v1} (backed up in \texttt{backups/v1/})
    \item \textbf{Re-extracted} (improved Potency templates)
    \item \textbf{Orthogonalised} (Gram--Schmidt post-processing)
\end{enumerate}

\bigskip

\textbf{Metrics:}
\begin{itemize}
    \item \textbf{Cross-dimension entanglement:} Mean $|\cos(\mathbf{d}_i, \mathbf{d}_j)|$ per layer.
          Target: $< 0.1$ after orthogonalisation
    \item \textbf{Within-dimension cosine:} How much did directions change
          from v1 $\rightarrow$ v2?
    \item \textbf{Reader quality:} Compare $\rho$, $R^2$, MAE for each
          direction set
\end{itemize}
\end{frame}

% -----------------------------------------------------------------------------
\begin{frame}{Exp 10: Statistical Significance}

\textbf{Purpose:} Post-hoc analysis aggregating results from Exps 02, 03,
and 08 into publication-quality statistical tests.

\bigskip

\textbf{Tests performed:}
\begin{itemize}
    \item \textbf{Bootstrap CIs} (10{,}000 resamples) on hit rates and
          mean distance improvements
    \item \textbf{Wilcoxon signed-rank} test for paired comparisons
    \item \textbf{Paired permutation test} (10{,}000 permutations) as
          nonparametric alternative
    \item \textbf{Cohen's $d$} (paired) for effect sizes
\end{itemize}

\bigskip

\textbf{Outputs:}
\begin{itemize}
    \item JSON with all pairwise test statistics
    \item LaTeX table suitable for direct inclusion in the AAAI paper
    \item Forest plot of effect sizes colour-coded by significance level
\end{itemize}
\end{frame}

\begin{frame}{Known Limitations}

\begin{enumerate}
    \item \textbf{Potency direction is broken.}
    The RepE-extracted Potency direction does not reliably encode P.
    The coefficient sweep confirms it acts in the \emph{wrong direction}.
    The reader $\rho = 0.57$ is too low for closed-loop control.

    \item \textbf{Prompt engineering outperforms RepE on 2/3 dimensions.}
    For P and A, explicit affective instructions in the system prompt
    achieve lower distance-to-target than activation steering.

    \item \textbf{Template-based calibration data.}
    Current calibration uses template sentences (``A person who \{verb\}
    another person''). LLM-generated utterances would yield more natural
    evaluation data and better calibration.

    \item \textbf{Single model.}
    All experiments run on Llama-3.1-8B-Instruct. Generalisability to
    other architectures / scales is unknown.

    \item \textbf{Human evaluation is missing.}
    No human judges have rated whether steered responses \emph{feel}
    affectively different. Current metrics are all automated.
\end{enumerate}
\end{frame}
